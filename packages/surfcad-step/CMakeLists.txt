cmake_minimum_required(VERSION 3.20)
project(surfcad_step LANGUAGES C CXX)

# STEP reader and mesher for the browser, built from Open CASCADE Technology
# (OCCT). OCCT is LGPL-2.1 with the Open CASCADE exception, so this module is
# kept apart from the rest of the app: it is its own wasm file, loaded by URL
# at run time, and everything needed to rebuild or replace it is in this
# package and in scripts/step/build-step-wasm.mjs. See LICENSES.md.
#
# OCCT's own CMake project is not used. It builds whole toolkits and their
# dependency closure (TKDESTEP pulls the document framework and the
# visualization toolkits). This file compiles the sources of the toolkits
# listed below into one static archive and lets the linker keep the objects
# the wrapper reaches.

if(NOT OCCT_DIR)
    message(FATAL_ERROR "OCCT_DIR is required (Open-Cascade-SAS/OCCT checkout at the pinned tag)")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

# Stated here so the optimisation level does not depend on the toolchain
# file's Release default. The link step reads it too (wasm-opt level).
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

# Toolkits whose sources are compiled.
set(SURFCAD_STEP_TOOLKITS
    TKernel
    TKMath
    TKG2d
    TKG3d
    TKGeomBase
    TKBRep
    TKGeomAlgo
    TKTopAlgo
    TKShHealing
    TKMesh
    TKXSBase
    TKDE
    TKDESTEP
)

# Toolkits that only lend headers. Some compiled sources include document
# framework headers for types they hold by handle. None of these toolkits is
# compiled, so a call into one of them fails the link.
set(SURFCAD_STEP_HEADER_TOOLKITS
    TKCDF
    TKLCAF
    TKCAF
    TKVCAF
    TKXCAF
    TKService
    TKV3d
)

# Packages and files that need the document framework (XCAF) at link time.
# The wrapper reads with STEPControl_Reader, which does not.
set(SURFCAD_STEP_SKIP_PACKAGES STEPCAFControl)
set(SURFCAD_STEP_SKIP_FILES
    DESTEP_Provider.cxx
    DESTEP_ConfigurationNode.cxx
    TKDESTEP.cxx
)

set(OCCT_INC_DIR "${CMAKE_BINARY_DIR}/occt-inc")
file(MAKE_DIRECTORY "${OCCT_INC_DIR}")

function(surfcad_step_toolkit_dir toolkit out_var)
    file(GLOB hits LIST_DIRECTORIES true "${OCCT_DIR}/src/*/${toolkit}")
    list(LENGTH hits count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "toolkit ${toolkit} was not found exactly once under ${OCCT_DIR}/src")
    endif()
    list(GET hits 0 hit)
    set(${out_var} "${hit}" PARENT_SCOPE)
endfunction()

set(OCCT_SOURCES)
foreach(toolkit IN LISTS SURFCAD_STEP_TOOLKITS SURFCAD_STEP_HEADER_TOOLKITS)
    surfcad_step_toolkit_dir(${toolkit} toolkit_dir)
    include("${toolkit_dir}/PACKAGES.cmake")
    list(FIND SURFCAD_STEP_TOOLKITS ${toolkit} compiled_index)
    foreach(package IN LISTS OCCT_${toolkit}_LIST_OF_PACKAGES)
        include("${toolkit_dir}/${package}/FILES.cmake")
        list(FIND SURFCAD_STEP_SKIP_PACKAGES ${package} skip_package)
        foreach(name IN LISTS OCCT_${package}_FILES)
            set(path "${toolkit_dir}/${package}/${name}")
            if(name MATCHES "\\.(hxx|lxx|gxx|pxx|h|hpp)$")
                configure_file("${path}" "${OCCT_INC_DIR}/${name}" COPYONLY)
            elseif(name MATCHES "\\.(cxx|cpp|c)$" AND compiled_index GREATER -1 AND skip_package EQUAL -1)
                list(FIND SURFCAD_STEP_SKIP_FILES ${name} skip_file)
                if(skip_file EQUAL -1)
                    list(APPEND OCCT_SOURCES "${path}")
                endif()
            endif()
        endforeach()
    endforeach()
endforeach()

# Standard_Version.hxx is generated by OCCT's CMake. The date is fixed so the
# build is reproducible.
include("${OCCT_DIR}/adm/cmake/version.cmake")
set(OCCT_VERSION_DATE "1970-01-01")
set(SET_OCC_VERSION_DEVELOPMENT "")
configure_file("${OCCT_DIR}/adm/templates/Standard_Version.hxx.in" "${OCCT_INC_DIR}/Standard_Version.hxx" @ONLY)

add_library(occt STATIC ${OCCT_SOURCES})
target_include_directories(occt SYSTEM PUBLIC "${OCCT_INC_DIR}")
# No_Exception matches OCCT's release default (BUILD_RELEASE_DISABLE_EXCEPTIONS):
# it drops the range and null checks behind Standard_*_Raise_if. Real failures
# still throw. OCC_CONVERT_SIGNALS is left out: wasm has no signals.
target_compile_definitions(occt PUBLIC No_Exception)
target_compile_options(occt PRIVATE -w)

add_executable(surfcad_step src/api.cpp)
target_link_libraries(surfcad_step PRIVATE occt)

if(EMSCRIPTEN)
    # Native wasm exceptions. OCCT throws on ordinary control paths, and the
    # JS-based fallback (-fexceptions) is about twice as slow here.
    target_compile_options(occt PUBLIC -fwasm-exceptions)
    target_link_options(surfcad_step PRIVATE
        "-fwasm-exceptions"
        "-sWASM_LEGACY_EXCEPTIONS=1"
        "-sALLOW_MEMORY_GROWTH=1"
        "-sINITIAL_MEMORY=67108864"
        "-sMAXIMUM_MEMORY=4294967296"
        "-sSTACK_SIZE=16777216"
        "-sENVIRONMENT=web,webview,worker,node"
        "-sMODULARIZE=1"
        "-sEXPORT_ES6=1"
        "-sEXPORT_NAME=createSurfcadStep"
        "-sEXPORTED_FUNCTIONS=_surfcad_step_mesh,_surfcad_step_free,_surfcad_step_abi,_malloc,_free"
        "-sEXPORTED_RUNTIME_METHODS=HEAPU8"
        "-sFILESYSTEM=0"
        "-sNO_EXIT_RUNTIME=1"
        "-sINVOKE_RUN=0"
        "-sALLOW_UNIMPLEMENTED_SYSCALLS=1"
        "-sERROR_ON_UNDEFINED_SYMBOLS=1"
    )
endif()
